###cmd command for creating hardware fingerprinting
powershell.exe -NoProfile -ExecutionPolicy Bypass -Command "& { try { $sys_uuid = ''; try { $wmicOut = (wmic csproduct get uuid 2>$null); if ($wmicOut) { foreach ($l in ($wmicOut -split '`r?`n')) { $t = $l.Trim(); if ($t -and ($t.ToUpper() -notmatch 'UUID')) { $sys_uuid = $t; break } } } } catch {}; if (-not $sys_uuid) { try { $sys_uuid = (Get-CimInstance -Class Win32_ComputerSystemProduct -ErrorAction SilentlyContinue).UUID; if ($sys_uuid) { $sys_uuid = $sys_uuid.Trim() } } catch {} }; if (-not $sys_uuid) { $sys_uuid = '' }; $mac = '00:00:00:00:00:00'; $gotMac = $false; try { $def = 'using System; using System.Runtime.InteropServices; public class WNet { [DllImport(\"rpcrt4.dll\")] public static extern int UuidCreateSequential(out Guid guid); }'; Add-Type -TypeDefinition $def -ErrorAction SilentlyContinue; $guid = [Guid]::Empty; [WNet]::UuidCreateSequential([ref]$guid) | Out-Null; $b = $guid.ToByteArray(); if ($b.Length -ge 16) { $mac = [string]::Format('{0:x2}:{1:x2}:{2:x2}:{3:x2}:{4:x2}:{5:x2}', $b[10], $b[11], $b[12], $b[13], $b[14], $b[15]); $gotMac = $true } } catch {}; if (-not $gotMac) { try { $net = (Get-CimInstance Win32_NetworkAdapter | Where-Object { $_.PhysicalAdapter -and $_.MACAddress } | Sort-Object Index | Select-Object -First 1); if ($net) { $mac = $net.MACAddress.ToLower().Trim().Replace('-', ':') } } catch {} }; $cpu_arch = if ($env:PROCESSOR_ARCHITEW6432) { $env:PROCESSOR_ARCHITEW6432 } else { $env:PROCESSOR_ARCHITECTURE }; if (-not $cpu_arch) { $cpu_arch = '' }; $processor = if ($env:PROCESSOR_IDENTIFIER) { $env:PROCESSOR_IDENTIFIER.Trim() } else { '' }; $raw = 'spud::' + $sys_uuid + '::' + $mac + '::' + $cpu_arch + '::' + $processor; $hasher = [System.Security.Cryptography.SHA256]::Create(); $bytes = [System.Text.Encoding]::UTF8.GetBytes($raw); $sha = [System.BitConverter]::ToString($hasher.ComputeHash($bytes)).Replace('-', '').ToUpper(); $hwid = 'SPUD-' + $sha.Substring(0,4) + '-' + $sha.Substring(4,4) + '-' + $sha.Substring(8,4) + '-' + $sha.Substring(12,4); Write-Host ''; Write-Host '=======================================================' -ForegroundColor Cyan; Write-Host '       SpudManager Unique Machine Fingerprint          ' -ForegroundColor Cyan; Write-Host '=======================================================' -ForegroundColor Cyan; Write-Host ('Machine ID: ' + $hwid) -ForegroundColor Green; Write-Host '-------------------------------------------------------'; Write-Host 'Add this to your .env file or system environment variables:'; Write-Host ('ALLOWED_MACHINE_IDS=\"' + $hwid + '\"') -ForegroundColor Yellow; Write-Host '=======================================================' -ForegroundColor Cyan; Write-Host ''; try { Set-Clipboard -Value $hwid -ErrorAction SilentlyContinue; Write-Host ' [Copied to clipboard automatically!] ' -ForegroundColor DarkGreen } catch { try { $hwid | clip.exe } catch {} } } catch { Write-Host $_.Exception.Message -ForegroundColor Red } }"
